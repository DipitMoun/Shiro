/**
 * Work-level chat: one Matrix room per user + work, one thread per episode.
 *
 * The model opens only the user's thread and switches it when the episode rail
 * changes. Timeline state is cached by the repository, so navigation does not
 * restart the conversation from an empty list.
 */
class SocialChatScreenModel(
    private val roomService: SocialRoomService = Injekt.get(),
    private val session: MatrixSessionManager = Injekt.get(),
    timelineRepo: SocialThreadRepository = Injekt.get(),
    dispatcher: SocialCommandDispatcher = Injekt.get(),
    push: PushRegistrar = Injekt.get(),
    private val catalog: SocialThreadCatalog = Injekt.get(),
    private val avatarLoader: SocialAvatarLoader = Injekt.get(),
    private val friendStore: SocialFriendStore = Injekt.get(),
    private val inviteAccepter: SocialFriendInviteAccepter = Injekt.get(),
) : ScreenModel {

    private val controller = SocialOverlayController(
        session = session,
        timelineRepo = timelineRepo,
        dispatcher = dispatcher,
        push = push,
        scope = screenModelScope,
    )

    val state = controller.state

    private val _friends = MutableStateFlow<List<SocialChatFriend>>(emptyList())
    val friends: StateFlow<List<SocialChatFriend>> = _friends.asStateFlow()

    private val _selfAvatar = MutableStateFlow<ByteArray?>(null)
    val selfAvatar: StateFlow<ByteArray?> = _selfAvatar.asStateFlow()

    private var ensuredRoomForWork = false

    init {
        controller.connect()
        screenModelScope.launch {
            if (session.state.value is SocialSessionState.Unauthenticated) {
                controller.restore()
            }
            _selfAvatar.value = avatarLoader.loadOwnAvatar()
            inviteAccepter.acceptInvitesFromFriends()
        }
    }

    fun selectEpisode(
        context: SocialChatContext,
        selfMxid: String,
        selfDisplayName: String?,
        episode: SocialChatEpisode,
    ) {
        screenModelScope.launch {
            controller.prepareWork()
            if (!ensureRoom(context, selfMxid, selfDisplayName)) return@launch
            openEpisode(context, selfMxid, selfDisplayName, episode)
            refreshFriends(context, selfMxid, episode)
        }
    }

    fun send(
        text: String,
        positions: List<Long>,
        episode: SocialChatEpisode,
        context: SocialChatContext,
        inReplyToEventId: String? = null,
    ) {
        val body = text.trim()
        if (body.isEmpty() && positions.isEmpty()) return
        if (state.value.thread == null) return
        screenModelScope.launch {
            if (positions.isEmpty()) {
                controller.reply(body, inReplyToEventId)
            } else {
                controller.publish(
                    buildMomentPayload(
                        context = context,
                        episode = episode,
                        note = body,
                        positions = positions,
                    ),
                )
            }
        }
    }

    fun react(eventId: String, key: String) {
        if (state.value.thread == null) return
        screenModelScope.launch { controller.react(eventId, key) }
    }

    private suspend fun refreshFriends(
        context: SocialChatContext,
        selfMxid: String,
        episode: SocialChatEpisode,
    ) {
        if (_selfAvatar.value == null) {
            _selfAvatar.value = avatarLoader.loadOwnAvatar()
        }
        val discovered = when (val result = catalog.discover(context.work, episode.key, selfMxid)) {
            is SocialResult.Success -> result.value.associateBy { it.ref.authorMxid }
            is SocialResult.Failure -> emptyMap()
        }
        val friends = mutableListOf<SocialChatFriend>()
        friendStore.friends.value.forEach { record ->
            val thread = discovered[record.mxid]
            val name = record.displayName
                ?: thread?.displayName
                ?: record.mxid.substringBefore(':').removePrefix("@")
            val avatar = avatarLoader.loadProfileAvatar(record.mxid)
                ?: thread?.avatarUrl?.let { avatarLoader.loadAvatar(it) }
            friends += SocialChatFriend(
                mxid = record.mxid,
                name = name,
                avatarBytes = avatar,
            )
        }
        _friends.value = friends
    }

    private fun buildMomentPayload(
        context: SocialChatContext,
        episode: SocialChatEpisode,
        note: String,
        positions: List<Long>,
    ): MomentPayload {
        val durationMs = episode.durationSeconds.takeIf { it > 0L }?.times(1000L)
        return MomentPayload(
            v = MomentPayload.VERSION,
            work = MomentWork(
                id = context.work.id.value,
                ids = context.work.ids,
                title = context.work.title,
            ),
            episode = MomentEpisode(
                key = episode.key.value,
                number = episode.number,
                season = null,
                kind = "regular",
                numbering = "unknown",
                title = episode.title,
            ),
            moments = positions
                .distinct()
                .take(5)
                .map { positionMs ->
                    MomentAnchor(
                        positionMs = positionMs.coerceAtLeast(0L),
                        beforeMs = 15_000L,
                        afterMs = 15_000L,
                    )
                },
            note = note.takeIf { it.isNotBlank() },
            source = MomentSource(
                key = context.work.id.value,
                kind = "local",
                durationMs = durationMs,
            ),
            capturedAt = Instant.now().toString(),
            client = MomentClient("Shiro", BuildConfig.VERSION_NAME),
        )
    }

    private suspend fun ensureRoom(
        context: SocialChatContext,
        selfMxid: String,
        selfDisplayName: String?,
    ): Boolean {
        if (ensuredRoomForWork) return true
        val result = roomService.ensureRoom(
            authorMxid = selfMxid,
            authorDisplayName = selfDisplayName,
            selfMxid = selfMxid,
            work = context.work,
            friends = friendStore.asAudienceFriends(),
        )
        return when (result) {
            is SocialResult.Success -> {
                ensuredRoomForWork = true
                true
            }
            is SocialResult.Failure -> {
                controller.reportError(result.error)
                false
            }
        }
    }

    private suspend fun openEpisode(
        context: SocialChatContext,
        selfMxid: String,
        selfDisplayName: String?,
        episode: SocialChatEpisode,
    ) {
        val roomKey = SocialRoomKey(authorMxid = selfMxid, workId = context.work.id)
        when (val root = roomService.ensureThreadRoot(roomKey, episode.key.value, context.title)) {
            is SocialResult.Success -> {
                controller.openOwnWork(
                    work = context.work,
                    episodeKey = episode.key,
                    selfMxid = selfMxid,
                    selfDisplayName = selfDisplayName,
                )
            }
            is SocialResult.Failure -> controller.reportError(root.error)
        }
    }
}

private val chatTimeFormatter: DateTimeFormatter =
    DateTimeFormatter.ofPattern("HH:mm").withZone(ZoneId.systemDefault())

private val defaultReactions = listOf("ðŸ‘", "â¤ï¸", "ðŸ˜‚", "ðŸ˜®", "ðŸ˜¢", "ðŸ”¥")
private const val REACTION_PREFS = "shiro_social_reactions"
private const val REACTION_KEY = "recent"
private const val REACTION_SEPARATOR = "\u001F"

private fun loadRecentReactions(context: Context): List<String> {
    val raw = context.getSharedPreferences(REACTION_PREFS, Context.MODE_PRIVATE)
        .getString(REACTION_KEY, "")
        .orEmpty()
    return raw.split(REACTION_SEPARATOR).filter { it.isNotBlank() }
}

private fun rememberRecentReaction(context: Context, key: String): List<String> {
    val next = (listOf(key) + loadRecentReactions(context) + defaultReactions)
        .distinct()
        .take(18)
    context.getSharedPreferences(REACTION_PREFS, Context.MODE_PRIVATE)
        .edit()
        .putString(REACTION_KEY, next.joinToString(REACTION_SEPARATOR))
        .apply()
    return next
}

private fun messageText(item: SocialTimelineItem): String = when (item) {

@Composable
private fun ChatMessage(
    item: SocialTimelineItem,
    selfMxid: String,
    participants: Map<String, SocialChatFriend>,
    messageById: Map<String, SocialTimelineItem>,
    currentEpisode: SocialChatEpisode,
    episodes: List<SocialChatEpisode>,
    posterUrl: String?,
    showHeader: Boolean,
    onPreview: (SocialChatEpisode, Long, Boolean) -> Unit,
    onLongPress: (SocialTimelineItem) -> Unit,
    onReaction: (String) -> Unit,
) {
    val participant = participants[item.sender] ?: fallbackParticipant(item.sender)
    val momentEpisode = when (item) {
        is SocialTimelineItem.MomentItem -> {
            episodes.firstOrNull { it.key.value == item.payload.episode.key } ?: currentEpisode
        }
        is SocialTimelineItem.ReplyItem -> currentEpisode
    }
    val positions = when (item) {
        is SocialTimelineItem.MomentItem -> item.payload.moments.map { it.positionMs }
        is SocialTimelineItem.ReplyItem -> emptyList()
    }
    val replyTarget = (item as? SocialTimelineItem.ReplyItem)
        ?.inReplyToEventId
        ?.let(messageById::get)

    DiscordMessageRow(
        participant = participant,
        text = messageText(item),
        timestamp = item.sentAt,
        replyTarget = replyTarget,
        participants = participants,
        reactions = item.reactions.map { it.key to it.count },
        positions = positions,
        episode = momentEpisode,
        posterUrl = posterUrl,
        isMine = item.sender == selfMxid,
        showHeader = showHeader,
        onPreview = onPreview,
        onLongPress = { onLongPress(item) },
        onReaction = onReaction,
    )
}

@Composable
private fun TimecodePickerDialog(
    animeTitle: String,
    episode: SocialChatEpisode,
    onDismiss: () -> Unit,
    onPick: (Long) -> Unit,
) {
    val maxMinutes = (episode.durationSeconds / 60L).toInt().coerceAtLeast(0)
    var minute by remember(episode.key.value) { mutableIntStateOf(0) }
    var seconds by remember(episode.key.value) { mutableIntStateOf(0) }
    val maxSecondsForMinute = if (
        episode.durationSeconds > 0L &&
        minute == maxMinutes
    ) {
        (episode.durationSeconds % 60L).toInt()
    } else {
        59
    }
    val effectiveSeconds = seconds.coerceAtMost(maxSecondsForMinute)
    val pickedSeconds = (minute * 60L + effectiveSeconds).let { total ->
        if (episode.durationSeconds > 0L) total.coerceAtMost(episode.durationSeconds) else total
    }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = {
            Column {
                Text(animeTitle, maxLines = 1, overflow = TextOverflow.Ellipsis)
                Text(
                    text = "${episode.key.value} Â· ${formatTimecode(episode.durationSeconds * 1000L)}",
                    fontSize = 11.sp,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    maxLines = 1,
                )
            }
        },
        text = {
            Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                Text("Choisis la minute", fontSize = 11.sp)
                LazyColumn(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(160.dp),
                    verticalArrangement = Arrangement.spacedBy(2.dp),
                ) {
                    items((0..maxMinutes).toList()) { value ->
                        Surface(
                            color = if (value == minute) MaterialTheme.colorScheme.primaryContainer
                            else Color.Transparent,
                            shape = RoundedCornerShape(7.dp),
                            modifier = Modifier
                                .fillMaxWidth()
                                .clickable {
                                    minute = value
                                    val max = if (
                                        episode.durationSeconds > 0L &&
                                        value == maxMinutes
                                    ) {
                                        (episode.durationSeconds % 60L).toInt()
                                    } else {
                                        59
                                    }
                                    seconds = seconds.coerceAtMost(max)
                                },
                        ) {
                            Text(
                                text = "$value min",
                                fontSize = 12.sp,
                                modifier = Modifier.padding(horizontal = 10.dp, vertical = 7.dp),
                            )
                        }
                    }
